cmake_minimum_required(VERSION 3.24)
project(lumina_lya LANGUAGES C CXX)

option(LYA_ENABLE_CUDA "Build the CUDA backend" ON)
option(LYA_BUILD_TESTS "Build lightweight host-only tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(HDF5 1.10 REQUIRED COMPONENTS C)

add_library(lya_host STATIC src/config.cpp src/io.cpp src/merge.cpp src/backend_cpu.cpp)
target_include_directories(lya_host PUBLIC include ${HDF5_INCLUDE_DIRS})
target_link_libraries(lya_host PUBLIC ${HDF5_LIBRARIES})
target_compile_options(lya_host PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)

if(LYA_ENABLE_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 80 90)
    endif()
    enable_language(CUDA)
    add_library(lya_device STATIC src/backend_cuda.cu)
    target_include_directories(lya_device PUBLIC include)
    set_target_properties(lya_device PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
        CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
    target_compile_options(lya_device PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
else()
    add_library(lya_device STATIC src/backend_stub.cpp)
    target_include_directories(lya_device PUBLIC include)
endif()

add_executable(lumina_lya src/main.cpp)
target_link_libraries(lumina_lya PRIVATE lya_host lya_device)
target_compile_options(lumina_lya PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)

if(LYA_BUILD_TESTS)
    enable_testing()
    add_executable(test_numerics tests/test_numerics.cpp)
    target_include_directories(test_numerics PRIVATE include)
    target_compile_options(test_numerics PRIVATE -ffp-contract=off)
    add_test(NAME numerical_primitives COMMAND test_numerics)
    add_executable(numerics_probe tests/numerics_probe.cpp)
    target_include_directories(numerics_probe PRIVATE include)
    target_compile_options(numerics_probe PRIVATE -ffp-contract=off)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend.cpp")
        add_executable(test_backend tests/test_backend.cpp)
        target_link_libraries(test_backend PRIVATE lya_host)
        add_test(NAME native_cpu_backend COMMAND test_backend)
    endif()
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, scipy, h5py"
            RESULT_VARIABLE LYA_PYTHON_DEPS OUTPUT_QUIET ERROR_QUIET)
        if(LYA_PYTHON_DEPS EQUAL 0)
            add_test(NAME independent_numerics COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_numerics.py" $<TARGET_FILE:numerics_probe>)
            if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pipeline.py")
                add_test(NAME tiny_pipeline COMMAND "${Python3_EXECUTABLE}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pipeline.py" $<TARGET_FILE:lumina_lya>)
            endif()
            if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_merge.py")
                add_test(NAME merge_validation COMMAND "${Python3_EXECUTABLE}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_merge.py" $<TARGET_FILE:lumina_lya>)
            endif()
            add_test(NAME python_reference COMMAND "${Python3_EXECUTABLE}" -m unittest discover
                -s "${CMAKE_CURRENT_SOURCE_DIR}/../cpu/tests" -v)
        endif()
    endif()
endif()
